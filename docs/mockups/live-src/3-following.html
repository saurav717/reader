class Router(nn.Module):
    def __init__(self, d_model, n_experts, k=2):
        super().__init__()
        self.gate = nn.Linear(d_model, n_experts, bias=False)
        self.bias = nn.Parameter(torch.zeros(n_experts), requires_grad=False)
        self.k = k

    def forward(self, x):
        scores = self.gate(x)                       # (tokens, experts)
        # The bias picks the experts; the gate value stays the raw score.
        top = torch.topk(scores + self.bias, self.k, dim=-1).indices
        gate = torch.gather(scores.softmax(-1), -1, top)
        self.last_loads = torch.bincount(top.flatten(), minlength=scores.shape[-1]).float()
        self.last_loads /= self.last_loads.sum()
        return top, gate

    def extra_repr(self):
        return f"k={self.k}, experts={self.bias.numel()}"
