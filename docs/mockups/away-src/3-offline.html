def importance(model, batches):
    """Activation-based importance, Minitron §3.1: mean |a| over a calibration set."""
    scores = {}
    for name, layer in model.named_modules():
        if isinstance(layer, nn.Linear):
            scores[name] = torch.zeros(layer.out_features)
    with torch.no_grad():
        for batch in batches:
            for name, act in trace(model, batch):
                scores[name] += act.abs().mean(dim=(0, 1))
    return scores
