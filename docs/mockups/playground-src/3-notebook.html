Tiling the softmax

Compute attention a block at a time, keeping a running max m and sum ℓ so the full N×N matrix is never written. ¶ FlashAttention-2 §3.1 · Alg. 1

def flash_attn(q, k, v, block=128):
    m = torch.full(q.shape[:-1], -torch.inf, device=q.device)
    l = torch.zeros_like(m); o = torch.zeros_like(q)
    for j in range(0, k.shape[-2], block):
        s = q @ k[..., j:j+block, :].mT / q.shape[-1]**.5
        m_new = torch.maximum(m, s.amax(-1))   # running max, eq. (3)
        p = torch.exp(s - m_new[..., None])
        l = torch.exp(m - m_new) * l + p.sum(-1)
        o = torch.exp(m - m_new)[..., None] * o + p @ v[..., j:j+block, :]
        m = m_new
    return o / l[..., None]
